# Search Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add client-side full-text search in the header and tag filtering on the blog listing page.

**Architecture:** A static Astro endpoint generates `search.json` at build time from all post frontmatter and body text. The header gets a vanilla JS search field that lazily loads MiniSearch and queries the index on input. The blog listing gets tag-filter buttons that hide/show posts client-side via `data-tags` attributes and URL params.

**Tech Stack:** MiniSearch 7.x, Astro Content Collections API (`getCollection`), vanilla JS with dynamic `import()`, CSS custom properties from existing design system.

```typescript
// src/pages/search.json.ts
import type { APIRoute } from 'astro';
import { getCollection } from 'astro:content';

function stripMarkdown(text: string): string {
  return text
    .replace(/!\[.*?\]\(.*?\)/g, '')         // images first
    .replace(/\[([^\]]+)\]\([^)]+\)/g, '$1') // links → label
    .replace(/`{3}[\s\S]*?`{3}/gm, '')       // fenced code blocks
    .replace(/`[^`]+`/g, '')                 // inline code
    .replace(/^#{1,6}\s+/gm, '')             // headings
    .replace(/\*\*([^*]+)\*\*/g, '$1')       // bold
    .replace(/\*([^*]+)\*/g, '$1')           // italic
    .replace(/^[-*+]\s+/gm, '')              // unordered lists
    .replace(/^\d+\.\s+/gm, '')              // ordered lists
    .replace(/^>\s+/gm, '')                  // blockquotes
    .replace(/\n{2,}/g, ' ')
    .replace(/\n/g, ' ')
    .trim();
}

export const GET: APIRoute = async () => {
  const posts = await getCollection('blog');

  const index = posts.map((post) => ({
    id: post.id,
    url: `/blog/${post.id}/`,
    title: post.data.title,
    description: post.data.description,
    tags: post.data.tags.join(' '),
    body: stripMarkdown(post.body ?? ''),
  }));

  return new Response(JSON.stringify(index), {
    headers: { 'Content-Type': 'application/json' },
  });
};
```

- [ ] **Step 3: Add `<script>` tag at the bottom of the file (after `</style>`)**

```html
<script>
	let miniSearch: InstanceType<typeof import('minisearch').default> | null = null;
	let loaded = false;

	async function loadSearch() {
		if (loaded) return;
		loaded = true;
		const input = document.getElementById('search-input') as HTMLInputElement | null;
		try {
			const [{ default: MiniSearch }, data] = await Promise.all([
				import('minisearch'),
				fetch('/search.json').then((r) => r.json()),
			]);
			miniSearch = new MiniSearch({
				fields: ['title', 'description', 'tags', 'body'],
				storeFields: ['url', 'title', 'tags'],
				searchOptions: {
					boost: { title: 3, description: 2, tags: 2, body: 1 },
					prefix: true,
					fuzzy: 0.2,
				},
			});
			miniSearch.addAll(data);
		} catch {
			if (input) input.disabled = true;
		}
	}

	function renderResults(results: { url: string; title: string; tags: string }[]) {
		const list = document.getElementById('search-results') as HTMLUListElement | null;
		if (!list) return;
		list.innerHTML = '';
		if (results.length === 0) {
			const li = document.createElement('li');
			li.className = 'no-results';
			li.textContent = 'Ничего не найдено';
			list.appendChild(li);
		} else {
			results.slice(0, 6).forEach(({ url, title, tags }) => {
				const li = document.createElement('li');
				const a = document.createElement('a');
				a.href = url;
				const titleSpan = document.createElement('span');
				titleSpan.className = 'result-title';
				titleSpan.textContent = title;
				a.appendChild(titleSpan);
				if (tags) {
					const tagsSpan = document.createElement('span');
					tagsSpan.className = 'result-tags';
					tagsSpan.textContent = tags.split(' ').slice(0, 3).join(' · ');
					a.appendChild(tagsSpan);
				}
				li.appendChild(a);
				list.appendChild(li);
			});
		}
		list.hidden = false;
	}

	document.addEventListener('DOMContentLoaded', () => {
		const input = document.getElementById('search-input') as HTMLInputElement | null;
		const results = document.getElementById('search-results') as HTMLUListElement | null;
		if (!input || !results) return;

		input.addEventListener('focus', loadSearch);

		input.addEventListener('input', () => {
			const q = input.value.trim();
			if (q.length < 2) {
				results.hidden = true;
				return;
			}
			if (!miniSearch) return;
			renderResults(miniSearch.search(q) as { url: string; title: string; tags: string }[]);
		});

		input.addEventListener('keydown', (e) => {
			if (e.key === 'Escape') {
				results.hidden = true;
				input.blur();
			}
		});

		document.addEventListener('click', (e) => {
			const wrapper = document.querySelector('.search-wrapper');
			if (wrapper && !wrapper.contains(e.target as Node)) {
				results.hidden = true;
			}
		});
	});
</script>
```

```html
<script>
	function applyFilter(tag: string) {
		document.querySelectorAll<HTMLButtonElement>('.tag-btn').forEach((btn) => {
			btn.classList.toggle('active', btn.dataset.tag === tag);
		});
		document.querySelectorAll<HTMLLIElement>('#post-list li').forEach((li) => {
			if (!tag) {
				li.hidden = false;
			} else {
				const tags = li.dataset.tags ?? '';
				li.hidden = !tags.split(' ').includes(tag);
			}
		});
	}

	document.addEventListener('DOMContentLoaded', () => {
		const params = new URLSearchParams(window.location.search);
		applyFilter(params.get('tag') ?? '');

		document.querySelectorAll<HTMLButtonElement>('.tag-btn').forEach((btn) => {
			btn.addEventListener('click', () => {
				const tag = btn.dataset.tag ?? '';
				const url = tag
					? `${window.location.pathname}?tag=${tag}`
					: window.location.pathname;
				history.pushState({}, '', url);
				applyFilter(tag);
			});
		});
	});
</script>
```
